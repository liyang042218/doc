PHP 数组 ​

数组是 PHP 中重要的数据结构，可以在单个变量名中存储多个值。

索引数组 ​

创建索引数组 ​

php
<?php
// 使用 array() 函数
$fruits = array("苹果", "香蕉", "橙子");

// 使用短数组语法（PHP 5.4+）
$vegetables = ["胡萝卜", "西兰花", "菠菜"];

// 访问元素
echo $fruits[0]; // 苹果
echo $fruits[1]; // 香蕉

添加和修改元素 ​

php
<?php
$numbers = [1, 2, 3];

// 添加元素到末尾
$numbers[] = 4;

// 在指定索引添加元素
$numbers[10] = 5;

// 修改元素
$numbers[0] = 10;

print_r($numbers);

关联数组 ​

创建关联数组 ​

php
<?php
// 使用 array() 函数
$person = array(
    "name" => "张三",
    "email" => "zhangsan@example.com",
    "age" => 30
);

// 使用短数组语法
$user = [
    "username" => "zhangsan",
    "role" => "管理员",
    "active" => true
];

// 访问元素
echo $person["name"]; // 张三

遍历关联数组 ​

php
<?php
$user = [
    "username" => "zhangsan",
    "email" => "zhangsan@example.com",
    "role" => "管理员"
];

foreach ($user as $key => $value) {
    echo "$key: $value<br>";
}

多维数组 ​

二维数组 ​

php
<?php
$matrix = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
];

// 访问元素
echo $matrix[0][0]; // 1
echo $matrix[1][2]; // 6

// 遍历
for ($i = 0; $i < 3; $i++) {
    for ($j = 0; $j < 3; $j++) {
        echo $matrix[$i][$j] . " ";
    }
    echo "<br>";
}

关联多维数组 ​

php
<?php
$students = [
    [
        "name" => "张三",
        "grades" => [90, 85, 88]
    ],
    [
        "name" => "李四",
        "grades" => [95, 92, 89]
    ]
];

echo $students[0]["name"]; // 张三
echo $students[1]["grades"][1]; // 92

数组函数 ​

添加和删除元素 ​

php
<?php
$numbers = [1, 2, 3];

// 添加到末尾
array_push($numbers, 4, 5);

// 从末尾删除
array_pop($numbers);

// 添加到开头
array_unshift($numbers, 0);

// 从开头删除
array_shift($numbers);

print_r($numbers);

搜索数组 ​

php
<?php
$fruits = ["苹果", "香蕉", "橙子", "苹果"];

// 检查值是否存在
in_array("香蕉", $fruits); // true

// 获取值的第一个索引
array_search("苹果", $fruits); // 0

// 检查键是否存在
$person = ["name" => "张三", "age" => 30];
array_key_exists("name", $person); // true

排序数组 ​

php
<?php
$numbers = [3, 1, 4, 1, 5, 9, 2, 6];

// 升序排序
sort($numbers);
print_r($numbers); // [1, 1, 2, 3, 4, 5, 6, 9]

// 降序排序
rsort($numbers);
print_r($numbers); // [9, 6, 5, 4, 3, 2, 1, 1]

// 关联数组排序
$ages = ["张三" => 30, "李四" => 25, "王五" => 40];
asort($ages); // 按值排序
ksort($ages); // 按键排序

数组转换 ​

php
<?php
$numbers = [1, 2, 3, 4, 5];

// map - 转换每个元素
$squared = array_map(function($n) {
    return $n * $n;
}, $numbers);

// filter - 保留符合条件的元素
$evens = array_filter($numbers, function($n) {
    return $n % 2 == 0;
});

// reduce - 合并所有元素
$sum = array_reduce($numbers, function($carry, $n) {
    return $carry + $n;
}, 0);

print_r($squared); // [1, 4, 9, 16, 25]
print_r($evens);   // [2, 4]
echo $sum;         // 15

合并和切片 ​

php
<?php
$array1 = [1, 2, 3];
$array2 = [4, 5, 6];

// 合并数组
$merged = array_merge($array1, $array2);
print_r($merged); // [1, 2, 3, 4, 5, 6]

// 合并关联数组（后面的值会覆盖前面的）
$a1 = ["a" => 1, "b" => 2];
$a2 = ["b" => 3, "c" => 4];
$merged = array_merge($a1, $a2);

// 切片数组
$numbers = [1, 2, 3, 4, 5];
$slice = array_slice($numbers, 2, 2);
print_r($slice); // [3, 4]

常用数组函数 ​

php
<?php
$numbers = [1, 2, 3, 4, 5];

// 计算元素数量
count($numbers); // 5

// 检查数组是否为空
empty($numbers); // false

// 获取唯一值
$withDupes = [1, 2, 2, 3, 3, 3];
$unique = array_unique($withDupes);

// 反转数组
$reversed = array_reverse($numbers);

// 键值互换
$flipped = array_flip(["a" => 1, "b" => 2]);

数组解构（PHP 7.1+） ​

php
<?php
$colors = ["红色", "绿色", "蓝色"];

// list 赋值
list($first, $second, $third) = $colors;

// 简短语法
[$red, $green, $blue] = $colors;

echo $red;   // 红色
echo $green; // 绿色
echo $blue;  // 蓝色

// 跳过元素
[, , $blue] = $colors;

展开运算符（PHP 7.4+） ​

php
<?php
$arr1 = [1, 2, 3];
$arr2 = [4, 5, 6];

// 合并数组
$merged = [...$arr1, ...$arr2];

// 添加元素
$newArr = [0, ...$arr1, 7];

// 复制数组
$copy = [...$arr1];
