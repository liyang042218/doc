PHP 面向对象编程 ​

PHP 对面向对象编程有强大的支持，包括类、对象、继承、接口、Trait、命名空间等。

类和对象 ​

定义类 ​

php
<?php
class Person {
    // 属性
    public string $name;
    public int $age;

    // 构造函数
    public function __construct(string $name, int $age) {
        $this->name = $name;
        $this->age = $age;
    }

    // 方法
    public function greet(): string {
        return "你好，我的名字是 " . $this->name;
    }

    public function birthday(): void {
        $this->age++;
    }
}

// 创建对象
$person = new Person("张三", 30);
echo $person->greet(); // 你好，我的名字是 张三

属性类型 ​

php
<?php
class User {
    public string $name;        // 公有 - 任何地方都可访问
    protected string $email;     // 受保护 - 在类内部和子类中可访问
    private string $password;   // 私有 - 仅在当前类中可访问
    static int $userCount = 0;   // 静态 - 所有实例共享

    public function __construct(string $name, string $email, string $password) {
        $this->name = $name;
        $this->email = $email;
        $this->password = $password;
        self::$userCount++;
    }
}

类型声明和返回类型 ​

php
<?php
class Calculator {
    public function add(int $a, int $b): int {
        return $a + $b;

    public function divide(float $a, float $b): float {
        if ($b == 0) {

}

继承 ​

扩展类 ​

php
<?php
class Animal {
    protected string $name;

    public function __construct(string $name) {
        $this->name = $name;
    }

    public function speak(): string {
        return "发出某种声音";
    }
}

class Dog extends Animal {
    private string $breed;

    public function __construct(string $name, string $breed) {
        parent::__construct($name);
        $this->breed = $breed;
    }

    public function speak(): string {
        return $this->name . " 汪汪叫！";
    }

    public function getBreed(): string {
        return $this->breed;
    }
}

$dog = new Dog("旺财", "金毛");
echo $dog->speak(); // 旺财 汪汪叫！

抽象类 ​

php
<?php
abstract class Shape {
    abstract public function area(): float;
    abstract public function perimeter(): float;

    public function description(): string {
        return "这是一个形状";
    }
}

class Circle extends Shape {
    private float $radius;

    public function __construct(float $radius) {
        $this->radius = $radius;
    }

    public function area(): float {
        return pi() * $this->radius ** 2;
    }

    public function perimeter(): float {
        return 2 * pi() * $this->radius;
    }
}

class Rectangle extends Shape {
    private float $width;
    private float $height;

    public function __construct(float $width, float $height) {
        $this->width = $width;
        $this->height = $height;
    }

    public function area(): float {
        return $this->width * $this->height;
    }

    public function perimeter(): float {
        return 2 * ($this->width + $this->height);
    }
}

接口 ​

定义和实现接口 ​

php
<?php
interface Payable {
    public function getPaymentAmount(): float;
    public function getPaymentDate(): DateTime;
}

interface Discountable {
    public function applyDiscount(float $percentage): void;
    public function getDiscount(): float;
}

class Product implements Payable, Discountable {
    private string $name;
    private float $price;
    private float $discount = 0;

    public function __construct(string $name, float $price) {
        $this->name = $name;
        $this->price = $price;
    }

    public function getPaymentAmount(): float {
        return $this->price * (1 - $this->discount / 100);
    }

    public function getPaymentDate(): DateTime {
        return new DateTime();
    }

    public function applyDiscount(float $percentage): void {
        $this->discount = $percentage;
    }

    public function getDiscount(): float {
        return $this->discount;
    }
}

Trait ​

使用 Trait 实现代码复用 ​

php
<?php
trait Logger {
    private array $logs = [];

    public function log(string $message): void {
        $this->logs[] = date("Y-m-d H:i:s") . " - " . $message;
    }

    public function getLogs(): array {
        return $this->logs;
    }
}

trait Timestamps {
    private DateTime $createdAt;
    private DateTime $updatedAt;

    public function __construct() {
        $this->createdAt = new DateTime();
        $this->updatedAt = new DateTime();
    }

    public function touch(): void {
        $this->updatedAt = new DateTime();
    }
}

class User {
    use Logger, Timestamps;

    private string $name;

    public function __construct(string $name) {
        $this->name = $name;
        $this->log("用户已创建：$name");
    }
}

$user = new User("张三");
$user->log("用户登录");
print_r($user->getLogs());

静态方法和属性 ​

php
<?php
class Config {
    private static array $settings = [];
    private static ?Config $instance = null;

    private function __construct() {}

    public static function getInstance(): Config {
        if (self::$instance === null) {
            self::$instance = new Config();
        }
        return self::$instance;
    }

    public static function set(string $key, mixed $value): void {
        self::$settings[$key] = $value;
    }

    public static function get(string $key, mixed $default = null): mixed {
        return self::$settings[$key] ?? $default;
    }
}

Config::set("app_name", "我的应用");
Config::set("debug", true);

echo Config::get("app_name"); // 我的应用
echo Config::get("debug");    // 1

异常处理 ​

php
<?php
class ValidationException extends Exception {
    private array $errors;

    public function __construct(array $errors) {
        $this->errors = $errors;
        parent::__construct("验证失败");
    }

    public function getErrors(): array {
        return $this->errors;
    }
}

class UserValidator {
    public function validate(array $data): void {
        $errors = [];

        if (empty($data["name"])) {
            $errors[] = "用户名不能为空";
        }

        if (!filter_var($data["email"], FILTER_VALIDATE_EMAIL)) {
            $errors[] = "邮箱地址无效";
        }

        if (strlen($data["password"]) < 8) {
            $errors[] = "密码必须至少8个字符";
        }

        if (!empty($errors)) {
            throw new ValidationException($errors);
        }
    }
}

try {
    $validator = new UserValidator();
    $validator->validate([
        "name" => "",
        "email" => "无效邮箱",
        "password" => "短"
    ]);
} catch (ValidationException $e) {
    print_r($e->getErrors());
}

命名空间 ​

使用命名空间组织代码 ​

php
<?php
// 文件：src/Models/User.php
namespace App\Models;

class User {
    public string $name;
    public string $email;
}

// 文件：src/Services/UserService.php
namespace App\Services;

use App\Models\User;

class UserService {
    public function createUser(string $name, string $email): User {
        $user = new User();
        $user->name = $name;
        $user->email = $email;
        return $user;
    }
}
