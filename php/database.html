PHP 数据库操作 ​

PHP 提供了多种连接数据库的方式，包括 PDO（PHP Data Objects）和 MySQLi。

PDO 连接 ​

连接数据库 ​

php
<?php
$host = "localhost";
$dbname = "myapp";
$username = "root";
$password = "";

try {
    $pdo = new PDO(
        "mysql:host=$host;dbname=$dbname;charset=utf8mb4",
        $username,
        $password,
        [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_EMULATE_PREPARES => false
        ]
    );
    echo "连接成功";
} catch (PDOException $e) {
    echo "连接失败：" . $e->getMessage();
}

带 DSN 选项的连接 ​

php
<?php
$dsn = "mysql:host=localhost;dbname=myapp;charset=utf8mb4";

$options = [
    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES => false,
    PDO::MYSQL_ATTR_INIT_COMMAND => "SET NAMES utf8mb4"
];

$pdo = new PDO($dsn, "root", "", $options);

增删改查操作 ​

创建（INSERT） ​

php
<?php
// 单条插入
$sql = "INSERT INTO users (name, email, password) VALUES (:name, :email, :password)";
$stmt = $pdo->prepare($sql);

$stmt->execute([
    "name" => "张三",
    "email" => "zhangsan@example.com",
    "password" => password_hash("secret123", PASSWORD_DEFAULT)
]);

$lastInsertId = $pdo->lastInsertId();

// 批量插入
$users = [
    ["name" => "小明", "email" => "xiaoming@example.com"],
    ["name" => "小红", "email" => "xiaohong@example.com"]
];

$sql = "INSERT INTO users (name, email) VALUES (:name, :email)";
$stmt = $pdo->prepare($sql);

foreach ($users as $user) {
    $stmt->execute($user);
}

$affectedRows = $stmt->rowCount();

读取（SELECT） ​

php
<?php
// 获取单条记录
$sql = "SELECT * FROM users WHERE id = :id";
$stmt = $pdo->prepare($sql);
$stmt->execute(["id" => 1]);
$user = $stmt->fetch();

echo $user["name"];

// 获取所有记录
$sql = "SELECT * FROM users WHERE active = :active";
$stmt = $pdo->prepare($sql);
$stmt->execute(["active" => 1]);
$users = $stmt->fetchAll();

foreach ($users as $user) {
    echo $user["name"] . "<br>";
}

// 获取特定列
$sql = "SELECT name, email FROM users";
$stmt = $pdo->query($sql);
while ($row = $stmt->fetch()) {
    echo "{$row['name']} - {$row['email']}<br>";
}

更新（UPDATE） ​

php
<?php
$sql = "UPDATE users SET email = :email, updated_at = NOW() WHERE id = :id";
$stmt = $pdo->prepare($sql);

$stmt->execute([
    "email" => "newemail@example.com",
    "id" => 1
]);

$affectedRows = $stmt->rowCount();

删除（DELETE） ​

php
<?php
$sql = "DELETE FROM users WHERE id = :id";
$stmt = $pdo->prepare($sql);
$stmt->execute(["id" => 5]);

$affectedRows = $stmt->rowCount();

预处理语句 ​

为什么使用预处理语句？ ​

php
<?php
// 不安全（容易受到 SQL 注入攻击）
$sql = "SELECT * FROM users WHERE name = '$name'";

// 安全使用预处理语句
$sql = "SELECT * FROM users WHERE name = :name";
$stmt = $pdo->prepare($sql);
$stmt->execute(["name" => $name]);

参数绑定 ​

php
<?php
// 命名参数
$sql = "SELECT * FROM users WHERE name = :name AND age > :min_age";
$stmt = $pdo->prepare($sql);
$stmt->execute([
    "name" => "张三",
    "min_age" => 18
]);

// 位置参数
$sql = "INSERT INTO products (name, price) VALUES (?, ?)";
$stmt = $pdo->prepare($sql);
$stmt->bindValue(1, "小部件");
$stmt->bindValue(2, 19.99);
$stmt->execute();

// 通过引用绑定
$data = ["小工具", 29.99];
$stmt->bindParam(1, $data[0]);
$stmt->bindParam(2, $data[1]);
$stmt->execute();

事务 ​

php
<?php
try {
    $pdo->beginTransaction();

    $pdo->exec("INSERT INTO accounts (user_id, balance) VALUES (1, 1000)");
    $pdo->exec("INSERT INTO accounts (user_id, balance) VALUES (2, 500)");

    $pdo->exec("UPDATE accounts SET balance = balance - 100 WHERE user_id = 1");
    $pdo->exec("UPDATE accounts SET balance = balance + 100 WHERE user_id = 2");

    $pdo->commit();
    echo "事务执行成功";
} catch (Exception $e) {
    $pdo->rollBack();
    echo "事务失败：" . $e->getMessage();
}

MySQLi 方法 ​

MySQLi 连接 ​

php
<?php
$mysqli = new mysqli("localhost", "root", "", "myapp");

if ($mysqli->connect_error) {
    die("连接失败：" . $mysqli->connect_error);
}

$mysqli->set_charset("utf8mb4");

MySQLi 预处理语句 ​

php
<?php
// 预处理
$stmt = $mysqli->prepare("SELECT * FROM users WHERE email = ?");
$stmt->bind_param("s", $email);

// 执行
$email = "zhangsan@example.com";
$stmt->execute();

// 获取结果
$result = $stmt->get_result();
$user = $result->fetch_assoc();

// 多语句查询
$sql = "SELECT * FROM users; SELECT * FROM orders";
$mysqli->multi_query($sql);

查询操作 ​

LIKE 和通配符 ​

php
<?php
// 搜索以 '张' 开头的名字
$sql = "SELECT * FROM users WHERE name LIKE :pattern";
$stmt = $pdo->prepare($sql);
$stmt->execute(["pattern" => "张%"]);

// 搜索包含 '三' 的名字
$stmt->execute(["pattern" => "%三%"]);

IN 子句 ​

php
<?php
$ids = [1, 2, 3, 4, 5];
$placeholders = implode(",", array_fill(0, count($ids), "?"));

$sql = "SELECT * FROM users WHERE id IN ($placeholders)";
$stmt = $pdo->prepare($sql);
$stmt->execute($ids);
$users = $stmt->fetchAll();

JOIN 操作 ​

php
<?php
$sql = "
    SELECT orders.id, orders.order_date, users.name, users.email
    FROM orders
    INNER JOIN users ON orders.user_id = users.id
    WHERE orders.status = :status
";
$stmt = $pdo->prepare($sql);
$stmt->execute(["status" => "completed"]);
$orders = $stmt->fetchAll();

// LEFT JOIN 示例
$sql = "
    SELECT users.name, COUNT(orders.id) as order_count
    FROM users
    LEFT JOIN orders ON users.id = orders.user_id
    GROUP BY users.id
";

LIMIT 和 OFFSET ​

php
<?php
$sql = "SELECT * FROM users ORDER BY created_at DESC LIMIT :limit OFFSET :offset";
$stmt = $pdo->prepare($sql);
$stmt->bindValue("limit", 10, PDO::PARAM_INT);
$stmt->bindValue("offset", 20, PDO::PARAM_INT);
$stmt->execute();

$users = $stmt->fetchAll();

错误处理 ​

php
<?php
class DatabaseException extends Exception {
    private string $query;

    public function __construct(string $message, string $query = "") {
        parent::__construct($message);
        $this->query = $query;
    }

    public function getQuery(): string {
        return $this->query;
    }
}

function executeQuery(PDO $pdo, string $sql, array $params = []): PDOStatement {
    try {
        $stmt = $pdo->prepare($sql);
        $stmt->execute($params);
        return $stmt;
    } catch (PDOException $e) {
        throw new DatabaseException(
            "查询失败：" . $e->getMessage(),
            $sql
        );
    }
}

数据库辅助类 ​

php
<?php
class Database {
    private PDO $pdo;
    private static ?Database $instance = null;

    private function __construct() {
        $config = require "config.php";

        $this->pdo = new PDO(
            "mysql:host={$config['host']};dbname={$config['dbname']};charset=utf8mb4",
            $config["username"],
            $config["password"],
            [
                PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
            ]
        );
    }

    public static function getInstance(): Database {
        if (self::$instance === null) {
            self::$instance = new Database();
        }
        return self::$instance;
    }

    public function query(string $sql, array $params = []): PDOStatement {
        $stmt = $this->pdo->prepare($sql);
        $stmt->execute($params);
        return $stmt;
    }

    public function fetch(string $sql, array $params = []): ?array {
        $result = $this->query($sql, $params)->fetch();
        return $result ?: null;
    }

    public function fetchAll(string $sql, array $params = []): array {
        return $this->query($sql, $params)->fetchAll();
    }

    public function insert(string $table, array $data): int {
        $columns = implode(", ", array_keys($data));
        $placeholders = ":" . implode(", :", array_keys($data));

        $sql = "INSERT INTO $table ($columns) VALUES ($placeholders)";
        $this->query($sql, $data);

        return (int) $this->pdo->lastInsertId();
    }
}

// 使用示例
$db = Database::getInstance();
$user = $db->fetch("SELECT * FROM users WHERE id = :id", ["id" => 1]);
$users = $db->fetchAll("SELECT * FROM users WHERE active = :active", ["active" => 1]);
$id = $db->insert("users", ["name" => "张三", "email" => "zhangsan@example.com"]);
